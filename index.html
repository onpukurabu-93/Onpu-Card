// ポコン音を鳴らす関数 function playPokon() { // AudioContextの作成 const audioCtx = new (window.AudioContext || window.webkitAudioContext)(); // オシレーター（音の発生器）とゲイン（音量調整）を作成 const osc = audioCtx.createOscillator(); const gain = audioCtx.createGain(); osc.type = 'sine'; // まるみのある滑らかな波形 const now = audioCtx.currentTime; // 音の高さの変化（低めから一気に高く上がって落ちる「ポコン」の響き） osc.frequency.setValueAtTime(300, now); osc.frequency.exponentialRampToValueAtTime(800, now + 0.05); osc.frequency.exponentialRampToValueAtTime(150, now + 0.15); // 音量の変化（一瞬で鳴ってスッと消える設定） gain.gain.setValueAtTime(0.5, now); gain.gain.exponentialRampToValueAtTime(0.01, now + 0.15); // 接続して再生 osc.connect(gain); gain.connect(audioCtx.destination); osc.start(now); osc.stop(now + 0.15); }
